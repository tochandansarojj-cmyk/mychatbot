# ============================================================ # IRIS FLOWER CLASSIFICATION - MACHINE LEARNING + GRADIO UI # Google Colab - Complete Project in One Cell # Dataset: Scikit-learn built-in Iris dataset # Model: Random Forest Classifier # UI: Gradio # ============================================================ # 1. INSTALL REQUIRED LIBRARIES !pip install -q --upgrade gradio scikit-learn pandas matplotlib seaborn # 2. IMPORT LIBRARIES import gradio as gr import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns from sklearn.datasets import load_iris from sklearn.model_selection import train_test_split from sklearn.ensemble import RandomForestClassifier from sklearn.metrics import ( accuracy_score, classification_report, confusion_matrix ) # 3. LOAD DATASET FROM SCIKIT-LEARN LIBRARY iris = load_iris(as_frame=True) X = iris.data y = iris.target feature_names = list(X.columns) target_names = list(iris.target_names) # Dataset preview df = X.copy() df["Species"] = [target_names[i] for i in y] # 4. SPLIT DATASET X_train, X_test, y_train, y_test = train_test_split( X, y, test_size=0.20, random_state=42, stratify=y ) # 5. TRAIN MACHINE LEARNING MODEL model = RandomForestClassifier( n_estimators=150, max_depth=5, random_state=42 ) model.fit(X_train, y_train) # 6. MODEL EVALUATION y_pred = model.predict(X_test) accuracy = accuracy_score(y_test, y_pred) report = classification_report( y_test, y_pred, target_names=target_names, output_dict=True ) report_df = pd.DataFrame(report).transpose().round(3) cm = confusion_matrix(y_test, y_pred) # 7. PREDICTION FUNCTION def predict_flower( sepal_length, sepal_width, petal_length, petal_width ): input_data = pd.DataFrame( [[ sepal_length, sepal_width, petal_length, petal_width ]], columns=feature_names ) prediction = model.predict(input_data)[0] probabilities = model.predict_proba(input_data)[0] predicted_species = target_names[prediction] confidence = float(max(probabilities)) * 100 probability_dict = { target_names[i]: float(probabilities[i]) for i in range(len(target_names)) } result = f""" ### Prediction Result **Predicted Flower:** {predicted_species.title()} **Prediction Confidence:** {confidence:.2f}% **Input Measurements:** - Sepal Length: {sepal_length} cm - Sepal Width: {sepal_width} cm - Petal Length: {petal_length} cm - Petal Width: {petal_width} cm The prediction was generated using a trained Random Forest machine learning classifier. """ return predicted_species.title(), probability_dict, result # 8. CONFUSION MATRIX PLOT def create_confusion_matrix(): fig, ax = plt.subplots(figsize=(7, 5)) sns.heatmap( cm, annot=True, fmt="d", cmap="Blues", xticklabels=target_names, yticklabels=target_names, ax=ax ) ax.set_title("Model Confusion Matrix") ax.set_xlabel("Predicted Species") ax.set_ylabel("Actual Species") plt.tight_layout() return fig # 9. CUSTOM CSS custom_css = """ .gradio-container { font-family: Arial, sans-serif; } #main-title { text-align: center; } """ # 10. Algorithm: Random Forest Classifier - Number of Trees: 150 - Training Samples: {len(X_train)} - Testing Samples: {len(X_test)} - Total Dataset Samples: {len(df)} - Number of Classes: 3 ### Model Accuracy # {accuracy * 100:.2f}% """ ) gr.Markdown("### Classification Report") gr.Dataframe( value=report_df, label="Precision, Recall and F1 Score", interactive=False ) gr.Markdown("### Confusion Matrix") gr.Plot( value=create_confusion_matrix() ) with gr.Tab("📁 Dataset Information"): gr.Markdown( """ ## Iris Dataset This project uses the built-in Iris dataset from the Scikit-learn library. **Dataset Source:** https://scikit-learn.org/stable/modules/generated/sklearn.datasets.load_iris.html **Features:** 1. Sepal Length 2. Sepal Width 3. Petal Length 4. Petal Width **Target Classes:** - Setosa - Versicolor - Virginica """ ) gr.Markdown("### Dataset Preview") gr.Dataframe( value=df.head(20), label="First 20 Dataset Records", interactive=False ) gr.Markdown("### Dataset Statistics") gr.Dataframe( value=df.describe().round(2), interactive=False ) with gr.Tab("ℹ️ About Project"): gr.Markdown( """ ## About This Project This is an interactive machine learning application developed using Python. ### Machine Learning Workflow 1.
